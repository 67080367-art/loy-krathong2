      banana: { main: '#2d6a4f', rim: '#74c69d' },

      blue:   { back: '#90e0ef', front: '#0096c7' }

      marigold: { outer: '#ffb703', inner: '#fb8500' },
      lotus:    { outer: '#ff85a1', inner: '#ffe5ec' },
      rose:     { outer: '#e63946', inner: '#ffb3c1' }

    // ฟังก์ชันวาด SVG กระทงตามที่ผู้ใช้ตกแต่ง

      const b = baseColors[cfg.base] || baseColors.banana;
      const p = petalColors[cfg.petal] || petalColors.green;
      const f = flowerColors[cfg.flower] || flowerColors.marigold;
      const candleColor = cfg.candle || '#ffd166';

          <!-- ธูป -->

          <line x1="110" y1="105" x2="110" y2="48" stroke="#d4a373" stroke-width="3" stroke-linecap="round" />
          <line x1="124" y1="105" x2="136" y2="52" stroke="#d4a373" stroke-width="3" stroke-linecap="round" />
          <circle cx="84" cy="52" r="2.5" fill="#ff5400" />
          <circle cx="110" cy="48" r="2.5" fill="#ff5400" />
          <circle cx="136" cy="52" r="2.5" fill="#ff5400" />

    // อัปเดตพรีวิวในหน้าต่างตกแต่ง
    function updatePreview() {
      document.getElementById('krathongPreview').innerHTML = buildKrathongSVG(customConfig);
    }

    function setOpt(type, el) {
      el.parentElement.querySelectorAll('.selected').forEach(item => item.classList.remove('selected'));
      el.classList.add('selected');
      customConfig[type] = el.dataset.val;
      updatePreview();
    }

    function openModal() {
      updatePreview();
      modal.classList.add('active');
    }
    function closeModal() { modal.classList.remove('active'); }

    function escapeHTML(str) {
      return String(str).replace(/[&<>'"]/g, tag => ({
        '&': '&amp;', '<': '&lt;', '>': '&gt;', "'": '&#39;', '"': '&quot;'
      }[tag]));
    }

    // แสดงกระทงในแม่น้ำ
    const renderedIds = new Set();
    function renderKrathong(id, data, startNow = false) {
      if (renderedIds.has(id)) return;
      renderedIds.add(id);

      const wrapper = document.createElement('div');
      wrapper.className = 'krathong-wrapper';
      const topPercent = Math.random() * 62 + 8;
      const duration = Math.random() * 20 + 28;
      const scale = 0.75 + (topPercent / 100) * 0.4;

      wrapper.style.top = `${topPercent}%`;
      wrapper.style.setProperty('--duration', `${duration}s`);
      wrapper.style.transform = `scale(${scale})`;
      if (!startNow) wrapper.style.animationDelay = `-${Math.random() * duration}s, 0s`;

      wrapper.innerHTML = `
        <div class="wish-bubble">
          <span class="wish-name">${escapeHTML(data.name)}</span>
          <span class="wish-text">${escapeHTML(data.wish)}</span>
        </div>
        ${buildKrathongSVG(data.config || customConfig)}
      `;
      river.appendChild(wrapper);
    }

    // โหลดกระทงทั้งหมดจาก Firebase
    async function loadAllKrathongs() {
      if (FIREBASE_DB_URL.includes("YOUR_FIREBASE_URL")) {
        // กรณียังไม่ใส่ Firebase จะโชว์ตัวอย่าง
        renderKrathong('demo1', { name: 'เจ้าของเว็บ', wish: 'มาลอยกระทงด้วยกันนะ!', config: customConfig });
        return;
      }
      try {
        const res = await fetch(FIREBASE_DB_URL);
        const data = await res.json();
        if (data) {
          Object.entries(data).forEach(([id, item]) => renderKrathong(id, item, false));
        }
      } catch (e) { console.error(e); }
    }

    // ส่งกระทงใหม่ขึ้น Firebase
    async function submitKrathong() {
      const name = document.getElementById('userName').value.trim() || 'ผู้ไม่ประสงค์ออกนาม';
      const wish = document.getElementById('userWish').value.trim() || 'ขอให้มีความสุขสมหวัง';
      const payload = { name, wish, config: { ...customConfig }, createdAt: Date.now() };

      if (!FIREBASE_DB_URL.includes("YOUR_FIREBASE_URL")) {
        try {
          const res = await fetch(FIREBASE_DB_URL, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload)
          });
          const result = await res.json();
          renderKrathong(result.name, payload, true);
        } catch (e) { alert('ส่งข้อมูลไม่สำเร็จ ลองใหม่อีกครั้ง'); }
      } else {
        renderKrathong('local_' + Date.now(), payload, true);
      }

      document.getElementById('userName').value = '';
      document.getElementById('userWish').value = '';
      closeModal();
    }

    loadAllKrathongs();
    // เช็กกระทงใหม่ของเพื่อนๆ ทุก 5 วินาที
    setInterval(loadAllKrathongs, 5000);
